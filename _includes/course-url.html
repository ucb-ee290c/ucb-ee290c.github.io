{%- comment -%} Resolve a data-file URL. Absolute, mailto and #anchor pass through; a leading "/" gets site.baseurl; anything else stays relative to the current page. Usage: {% include course-url.html url=link.url %} {%- endcomment -%} {%- assign _u = include.url -%} {%- assign _first = _u | slice: 0 -%} {%- if _u contains '://' or _u contains 'mailto:' or _first == '#' -%} {{ _u }} {%- elsif _first == '/' -%} {{ _u | relative_url }} {%- else -%} {{ _u }} {%- endif -%}
